"""Candidate 1: fused stable row bitonic; gated experiment, not qualified/published.

No vendor sort on supported inputs. Whole values/indices are produced anew on
all calls. Uses the pinned FlagTree standard.py XOR partner-exchange primitive,
not an unsupported tuple/comparator tl.sort API. See README.md for boundaries.
"""
import operator
import sys

import torch
import triton
import triton.language as tl
from triton.language.standard import xor_sum

from flag_gems.runtime import torch_device_fn
from flag_gems.utils import libentry

IS_BASELINE = False
MAX_ROW_LENGTH = 4096
_BASELINE_STABLE = None


@triton.jit
def _indicator(LOG_BLOCK: tl.constexpr, BIT: tl.constexpr):
    # Same hypercube coordinate construction as pinned standard.py:366-370.
    return tl.reshape(tl.arange(0, 2), [1] * (LOG_BLOCK - BIT - 1) + [2] + [1] * BIT)


@triton.jit
def _partner_integer(bits, LOG_BLOCK: tl.constexpr, BIT: tl.constexpr):
    # Each axis has exactly 2 lanes: a ^ (a ^ b) = b, without float reduction,
    # overflow-prone arithmetic, conversion, or int64 truncation.
    return bits ^ xor_sum(bits, LOG_BLOCK - 1 - BIT, keep_dims=True)


@triton.jit
def _comes_before(a_bits, a_id, b_bits, b_id, N: tl.constexpr,
                  VALUE_DTYPE: tl.constexpr, IS_FLOAT: tl.constexpr,
                  IS_BOOL: tl.constexpr, DESCENDING: tl.constexpr):
    av = a_id < N
    bv = b_id < N
    if IS_BOOL:
        a = a_bits.to(tl.uint8)
        b = b_bits.to(tl.uint8)
    else:
        a = a_bits.to(VALUE_DTYPE, bitcast=True)
        b = b_bits.to(VALUE_DTYPE, bitcast=True)
    if IS_FLOAT:
        # Every half/bfloat value is exactly representable in float32; carrier
        # bits remain unchanged, including zeros and all NaN sign/payload bits.
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        an = a != a
        bn = b != b
        if DESCENDING:
            numeric_before = a > b
            nan_before = an & ~bn
        else:
            numeric_before = a < b
            nan_before = ~an & bn
        value_before = tl.where(an != bn, nan_before, numeric_before)
        equal = (a == b) | (an & bn)
    else:
        # VALUE_DTYPE is SIGNED tl.int64/int32 where appropriate; never cast
        # these values through floating point or subtract to compare.
        if DESCENDING:
            value_before = a > b
        else:
            value_before = a < b
        equal = a == b
    # Validity always precedes value direction. All equal values, both signed
    # zeros, and all NaNs keep ascending original id even in descending sort.
    tie = a_id < b_id
    ordered_valid = value_before | (equal & tie)
    return (av & ~bv) | ((av == bv) & tl.where(av, ordered_valid, tie))


@triton.jit
def _bitonic_stage(bits, ids, STAGE: tl.constexpr, LOG_BLOCK: tl.constexpr,
                   N: tl.constexpr, VALUE_DTYPE: tl.constexpr,
                   IS_FLOAT: tl.constexpr, IS_BOOL: tl.constexpr,
                   DESCENDING: tl.constexpr):
    # Compiler-only repair matching pinned _bitonic_merge_hypercube: a stage
    # is a JIT constexpr argument; pass STAGE-1-i DIRECTLY to constexpr BIT.
    # No repeatedly assigned constexpr local or Tensor-valued bit binding.
    if STAGE < LOG_BLOCK:
        flip = _indicator(LOG_BLOCK, STAGE)
    else:
        flip = tl.full((), 0, tl.int32)
    for i in tl.static_range(STAGE):
        other_bits = _partner_integer(bits, LOG_BLOCK, STAGE - 1 - i)
        other_ids = _partner_integer(ids, LOG_BLOCK, STAGE - 1 - i)
        is_right = _indicator(LOG_BLOCK, STAGE - 1 - i)
        other_before = _comes_before(other_bits, other_ids, bits, ids,
            N, VALUE_DTYPE, IS_FLOAT, IS_BOOL, DESCENDING)
        swap = other_before != ((flip ^ is_right) != 0)
        bits = tl.where(swap, other_bits, bits)
        ids = tl.where(swap, other_ids, ids)
    return bits, ids


@libentry()
@triton.jit
def _row_stable_bitonic(X, Y, INDEX,
                        N: tl.constexpr, BLOCK: tl.constexpr,
                        LOG_BLOCK: tl.constexpr,
                        VALUE_DTYPE: tl.constexpr, BITS_DTYPE: tl.constexpr,
                        IS_FLOAT: tl.constexpr, IS_BOOL: tl.constexpr,
                        DESCENDING: tl.constexpr,
                        ROW_SHAPE: tl.constexpr,
                        X_ROW_STRIDES: tl.constexpr, Y_ROW_STRIDES: tl.constexpr,
                        X_AXIS_STRIDE: tl.constexpr, Y_AXIS_STRIDE: tl.constexpr):
    row = tl.program_id(0).to(tl.int64)
    remaining = row
    xbase = tl.full((), 0, tl.int64)
    ybase = tl.full((), 0, tl.int64)
    # Decode non-sort coordinates, using 64-bit pointer arithmetic and the
    # actual view strides. Tensor pointer already includes storage_offset.
    for d in tl.static_range(len(ROW_SHAPE) - 1, -1, -1):
        coordinate = remaining % ROW_SHAPE[d]
        remaining = remaining // ROW_SHAPE[d]
        xbase += coordinate * X_ROW_STRIDES[d]
        ybase += coordinate * Y_ROW_STRIDES[d]
    col = tl.arange(0, BLOCK)
    present = col < N
    value = tl.load(X + xbase + col.to(tl.int64) * X_AXIS_STRIDE,
                    mask=present, other=0)
    if IS_BOOL:
        bits = value.to(tl.int8)
    else:
        bits = value.to(BITS_DTYPE, bitcast=True)
    ids = col
    if LOG_BLOCK > 0:
        bits = tl.reshape(bits, [2] * LOG_BLOCK)
        ids = tl.reshape(ids, [2] * LOG_BLOCK)
        # Semantic DESCENDING is inside comparator. The network's final order
        # stays forward, so validity and stable id ties are never reversed.
        for stage in tl.static_range(1, LOG_BLOCK + 1):
            bits, ids = _bitonic_stage(
                bits, ids, stage, LOG_BLOCK, N,
                VALUE_DTYPE, IS_FLOAT, IS_BOOL, DESCENDING)
        bits = tl.reshape(bits, [BLOCK])
        ids = tl.reshape(ids, [BLOCK])
    if IS_BOOL:
        sorted_value = bits.to(VALUE_DTYPE)
    else:
        sorted_value = bits.to(VALUE_DTYPE, bitcast=True)
    destination = ybase + col.to(tl.int64) * Y_AXIS_STRIDE
    tl.store(Y + destination, sorted_value, mask=present)
    tl.store(INDEX + destination, ids.to(tl.int64), mask=present)


def _fallback(inp, stable, dim, descending):
    if _BASELINE_STABLE is None:
        raise RuntimeError('sort candidate must be installed before use')
    # Calling saved stable ENTRY directly avoids original sort() looking up a
    # now-patched module-global sort_stable and accidentally recursing.
    return _BASELINE_STABLE(inp, stable=stable, dim=dim, descending=descending)


def _run(inp, stable, dim, descending):
    if not isinstance(inp, torch.Tensor):
        return _fallback(inp, stable, dim, descending)
    supported = (torch.float16, torch.bfloat16, torch.float32,
                 torch.int32, torch.int64, torch.bool)
    if (inp.dtype not in supported or inp.layout != torch.strided or
            inp.device.type not in ('flagos', 'privateuseone', 'cuda')):
        return _fallback(inp, stable, dim, descending)
    axis = operator.index(dim)
    rank = inp.ndim
    if rank == 0:
        if axis not in (-1, 0):
            raise IndexError('Dimension out of range for scalar sort')
        axis = 0
        n = 1
    else:
        if not -rank <= axis < rank:
            raise IndexError('Dimension out of range for sort')
        axis %= rank
        n = int(inp.shape[axis])
    if n > MAX_ROW_LENGTH or any(s < 0 for s in inp.stride()):
        return _fallback(inp, stable, dim, descending)
    rows = inp.numel() // n if n else 0
    if rows > 2147483647:
        return _fallback(inp, stable, dim, descending)
    # Fresh contiguous outputs with original logical shape; never return inp,
    # a view of inp, persistent buffers, prior answers, or a device/dtype copy.
    values = torch.empty(tuple(inp.shape), dtype=inp.dtype, device=inp.device)
    indices = torch.empty(tuple(inp.shape), dtype=torch.int64, device=inp.device)
    if inp.numel() == 0:
        return values, indices
    block = triton.next_power_of_2(n)
    log_block = block.bit_length() - 1
    row_shape = tuple(int(inp.shape[d]) for d in range(rank) if d != axis)
    xrows = tuple(int(inp.stride(d)) for d in range(rank) if d != axis)
    yrows = tuple(int(values.stride(d)) for d in range(rank) if d != axis)
    xs = int(inp.stride(axis)) if rank else 0
    ys = int(values.stride(axis)) if rank else 0
    if inp.dtype == torch.float16:
        value_dtype, bits_dtype = tl.float16, tl.int16
    elif inp.dtype == torch.bfloat16:
        value_dtype, bits_dtype = tl.bfloat16, tl.int16
    elif inp.dtype == torch.float32:
        value_dtype, bits_dtype = tl.float32, tl.int32
    elif inp.dtype == torch.int64:
        value_dtype, bits_dtype = tl.int64, tl.int64
    elif inp.dtype == torch.int32:
        value_dtype, bits_dtype = tl.int32, tl.int32
    else:
        value_dtype, bits_dtype = tl.int1, tl.int8
    with torch_device_fn.device(inp.device):
        _row_stable_bitonic[(rows,)](
            inp, values, indices, n, block, log_block,
            value_dtype, bits_dtype, inp.is_floating_point(), inp.dtype == torch.bool,
            bool(descending), row_shape, xrows, yrows, xs, ys,
            num_warps=1 if n <= 128 else 4)
    return values, indices


def sort(inp, dim=-1, descending=False):
    return _run(inp, False, dim, descending)


def sort_stable(inp, *, stable, dim=-1, descending=False):
    return _run(inp, stable, dim, descending)


def install(torch, flag_gems):
    """Install root + already loaded module aliases before first ATen sort call.

    Main's resident worker owns per-variant installation isolation. This helper
    is idempotent for the same imported candidate module, never mixes variants.
    """
    global _BASELINE_STABLE
    if flag_gems.sort_stable is sort_stable:
        return
    if _BASELINE_STABLE is not None:
        raise RuntimeError('Candidate already installed but root changed; use a fresh variant worker')
    _BASELINE_STABLE = flag_gems.sort_stable
    modules = {
        'flag_gems', 'flag_gems.ops', 'flag_gems.ops.sort',
        'flag_gems.runtime.backend._hygon.ops',
        'flag_gems.runtime.backend._hygon.ops.sort',
        getattr(_BASELINE_STABLE, '__module__', ''),
    }
    flag_gems.sort = sort
    flag_gems.sort_stable = sort_stable
    for name in modules:
        module = sys.modules.get(name)
        if module is not None:
            if hasattr(module, 'sort'):
                module.sort = sort
            if hasattr(module, 'sort_stable'):
                module.sort_stable = sort_stable
